cmake_minimum_required(VERSION 3.22)
project(CanvasSchwungProbe LANGUAGES C CXX)
set(BOXEDWINE_SOURCE "" CACHE PATH "Pinned BoxedWine source with the bridge overrides applied")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
    message(FATAL_ERROR "Use the Move Linux ARM64 cross toolchain")
endif()
if(BOXEDWINE_SOURCE)
    if(NOT EXISTS "${BOXEDWINE_SOURCE}/source/sdl/main.cpp")
        message(FATAL_ERROR "Set BOXEDWINE_SOURCE to the prepared pinned source")
    endif()
    include(BoxedWineLinux.cmake)
    add_canvas_linux_runtime("${BOXEDWINE_SOURCE}")
endif()
set(SCHWUNG_SDK_INCLUDE "" CACHE PATH "Directory containing Schwung's plugin_api_v1.h")
if(SCHWUNG_SDK_INCLUDE)
    add_library(CanvasSchwung MODULE Module.cpp smf.cpp)
    set_target_properties(CanvasSchwung PROPERTIES PREFIX "" OUTPUT_NAME "dsp" CXX_VISIBILITY_PRESET hidden)
    target_compile_features(CanvasSchwung PRIVATE cxx_std_20)
    target_include_directories(CanvasSchwung PRIVATE "${SCHWUNG_SDK_INCLUDE}")
    target_link_libraries(CanvasSchwung PRIVATE pthread dl)
endif()
if(NOT BOXEDWINE_SOURCE AND NOT SCHWUNG_SDK_INCLUDE)
    message(FATAL_ERROR "Set SCHWUNG_SDK_INCLUDE for the modules or BOXEDWINE_SOURCE for the compatibility worker")
endif()
