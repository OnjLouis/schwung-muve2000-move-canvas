cmake_minimum_required(VERSION 3.22)
project(Mu2000Schwung LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SMU_SOURCE "" CACHE PATH "Pinned S-MU2000 source")
set(SCHWUNG_INCLUDE "" CACHE PATH "Schwung host ABI include directory")
set(MU2000_RENDER_AHEAD 6 CACHE STRING "Schwung worker queue depth in 128-frame blocks")
set(MU2000_PLAYER_RENDER_AHEAD 12 CACHE STRING "MIDI player queue depth in 128-frame blocks")
option(MU2000_THREADED_ENGINE "Enable the firmware's per-sample slave thread" OFF)
option(MU2000_ABI_PROBE "Enable the private synchronous DSP comparison mode" OFF)
add_subdirectory("${SMU_SOURCE}" engine EXCLUDE_FROM_ALL)
include(../common/EngineSource.cmake)
file(READ "${CMAKE_CURRENT_BINARY_DIR}/PilotEngine.cpp" move_engine)
if(MU2000_THREADED_ENGINE)
    string(REPLACE "bool threaded = false;" "bool threaded = true;" move_engine "${move_engine}")
endif()
string(REPLACE "int native_engine = 0;" "int native_engine = 1;" move_engine "${move_engine}")
string(REPLACE "int native_fx = 0;" "int native_fx = 2;" move_engine "${move_engine}")
set(worker_start "\tmu->set_threaded(threaded);")
string(FIND "${move_engine}" "${worker_start}" worker_start_at)
if(worker_start_at EQUAL -1)
    message(FATAL_ERROR "Pinned S-MU2000 worker start changed; review the Move scheduler")
endif()
string(REPLACE "${worker_start}"
    "#if defined(__linux__) && defined(MU2000_PLAYER)\n\t{\n\t\t// Only the chip worker inherits this policy, not the boot thread's remaining work.\n\t\tmu_player::WorkerPolicy policy;\n\t\tmu->set_threaded(threaded);\n\t}\n#else\n${worker_start}\n#endif"
    move_engine "${move_engine}")
string(PREPEND move_engine "#include \"PlayerWorkerPolicy.h\"\n")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/MoveEngine.cpp" CONTENT "${move_engine}" @ONLY)
foreach(target mu2000 mu2000-midi-player)
    add_library(${target} MODULE Module.cpp "${CMAKE_CURRENT_BINARY_DIR}/MoveEngine.cpp" "${SMU_SOURCE}/src/xg/model.cpp")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${SMU_SOURCE}/src" "${SMU_SOURCE}/src/vst3" "${SCHWUNG_INCLUDE}")
    target_link_libraries(${target} PRIVATE smu2000_core ${CMAKE_DL_LIBS})
    if(target STREQUAL "mu2000-midi-player")
        target_compile_definitions(${target} PRIVATE MU2000_RENDER_AHEAD=${MU2000_PLAYER_RENDER_AHEAD})
    else()
        target_compile_definitions(${target} PRIVATE MU2000_RENDER_AHEAD=${MU2000_RENDER_AHEAD})
    endif()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(${target} PRIVATE "LINKER:-z,defs")
        # Zig links libc++ statically. Keep its process-exit callbacks mapped
        # when Schwung closes and later reopens the module.
        if(ZIG)
            target_link_options(${target} PRIVATE "LINKER:-z,nodelete")
        endif()
    endif()
    set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "dsp"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules/${target}")
endforeach()
target_compile_definitions(mu2000-midi-player PRIVATE MU2000_PLAYER=1)
if(MU2000_ABI_PROBE)
    target_compile_definitions(mu2000-midi-player PRIVATE MU2000_ABI_PROBE=1)
endif()
